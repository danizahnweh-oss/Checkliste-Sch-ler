    const SMILEYS = ['😞', '🙁', '😐', '🙂', '🤩'];

    const APP_ID = 'ki_checkliste_v3_full';


    function initChecklist() {

        const containers = document.querySelectorAll('.rating-row .smileys');

        

        containers.forEach((container) => {

            const cardId = container.closest('.item-card').dataset.id;

            

            SMILEYS.forEach((smiley, index) => {

                const radioId = `r_${cardId}_${index}`;

                const input = document.createElement('input');

                input.type = 'radio';

                input.name = `rating_${cardId}`;

                input.id = radioId;

                input.value = index + 1;

                input.dataset.persist = `radio_${cardId}`;


                const label = document.createElement('label');

                label.setAttribute('for', radioId);

                label.innerText = smiley;

                label.title = `Stufe ${index + 1}`;


                container.appendChild(input);

                container.appendChild(label);

            });

        });


        loadData();

        setupAutoSave();

    }


    function setupAutoSave() {

        document.getElementById('main-app').addEventListener('input', () => {

            const data = {};

            

            // Radios speichern

            document.querySelectorAll('input[type="radio"]:checked').forEach(radio => {

                data[radio.name] = radio.value;

            });


            // Textareas speichern

            document.querySelectorAll('.item-card').forEach(card => {

                const id = card.dataset.id;

                const text = card.querySelector('textarea').value;

                data[`text_${id}`] = text;

            });


            localStorage.setItem(APP_ID, JSON.stringify(data));

        });

    }


    function loadData() {

        const raw = localStorage.getItem(APP_ID);

        if (!raw) return;


        const data = JSON.parse(raw);


        // Radios wiederherstellen

        Object.keys(data).forEach(key => {

            if (key.startsWith('rating_')) {

                const val = data[key];

                const radio = document.querySelector(`input[name="${key}"][value="${val}"]`);

                if (radio) radio.checked = true;

            }

            if (key.startsWith('text_')) {

                const id = key.replace('text_', '');

                const card = document.querySelector(`.item-card[data-id="${id}"]`);

                if (card) {

                    card.querySelector('textarea').value = data[key];

                }

            }

        });

    }


    // Start

    window.onload = initChecklist;

</script>
